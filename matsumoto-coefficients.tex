\noindent
This note proves Theorem~1.1 of \texttt{mainv2} by a direct route. The Coxeter input is Tits's
word property and two facts about minimal coset representatives, all cited. Everything else is
stated for an arbitrary presheaf on a category $\Cx(W,S)$ built from a Coxeter system; precubical
sets enter only in \Cref{sec:cubes}. Locators for \cite{Paris} refer to arXiv:1211.7339v1.

\section{Coxeter background}

Let $(W,S)$ be a Coxeter system with length $\ell$ and Coxeter matrix $(m_{s,t})$. For $X \subseteq
S$, $W_X$ is the subgroup generated by $X$. Following \cite[\S2.2]{Paris}, an element $w \in W$ is
\textbf{$(\emptyset,X)$-minimal} if it is of minimal length in $wW_X$; write $W^X$ for the set of
these. For $s,t \in S$ and $k \in \N$, $\altw stk$ is the alternating word $sts\cdots$ of length $k$,
extending the notation of \cite[\S1]{Paris} (defined there for $k \ge 2$) to $k = 0, 1$, and also
the element of $W$ it represents. Put $\Sf := \{X \subseteq S : W_X
\text{ is finite}\}$.

\begin{theorem}[{\cite[Thm.~2.7]{Paris}}]\label{thm:tits}
  Let $w \in W$, and let $\mu, \mu'$ be two reduced expressions of $w$. Then there is a finite
  sequence of elementary $M$-transformations joining $\mu$ to $\mu'$.
\end{theorem}

Here an \textbf{elementary $M$-transformation} joins $\mu$ to $\mu'$ when there are words
$\nu_1,\nu_2$ and $s,t \in S$ with $m_{s,t} \neq \infty$, $\mu = \nu_1\,\altw st{m_{s,t}}\,\nu_2$ and
$\mu' = \nu_1\,\altw ts{m_{s,t}}\,\nu_2$ \cite[\S2.2]{Paris}. The same theorem is \cite[Thm.~3.3.1(ii)]{BB}.

\begin{proposition}[{\cite[Prop.~2.9]{Paris}}]\label{prop:min}
  Let $(W,S)$ be a Coxeter system.
  \begin{enumerate}
    \item[(2)] Let $X \subset S$, and let $w \in W$. Then $w$ is $(\emptyset,X)$-minimal if and
      only if $\ell(ws) > \ell(w)$ for all $s \in X$, and $\ell(ws) > \ell(w)$ for all $s \in X$ if
      and only if $\ell(wu) = \ell(w) + \ell(u)$ for all $u \in W_X$.
    \item[(4)] Let $X \subset S$, and let $w \in W_X$. If $\mu = s_1 \cdots s_\ell$ is a reduced
      expression of $w$, then $s_1, \dots, s_\ell \in X$.
  \end{enumerate}
\end{proposition}

\begin{lemma}\label{lem:prefix}
  If $x_1 \cdots x_k$ is a reduced expression, then so is $x_1 \cdots x_i$ for every $i \le k$.
\end{lemma}
\begin{proof}
  If $x_1\cdots x_i$ had an expression of length $< i$, appending $x_{i+1}\cdots x_k$ would give an
  expression of $x_1\cdots x_k$ of length $< k$.
\end{proof}

\begin{lemma}\label{lem:dihedral}
  Let $s \neq t$ in $S$ with $m := m_{s,t} < \infty$. Then $\altw stk$ is a reduced expression for
  every $k \le m$. Consequently, if $x$ is the $(j+1)$-st letter of $\altw stm$, then
  $\ell(\altw stj\, x) > \ell(\altw stj)$, so $\altw stj \in W^{\{x\}}$, for every $j < m$.
\end{lemma}
\begin{proof}
  Put $J = \{s,t\}$. By \cite[Prop.~2.4.1(i),(ii)]{BB}, $(W_J,J)$ is a Coxeter system whose length
  function is the restriction of $\ell$, and by the remark following that proposition its diagram is
  the single edge labelled $m$; so it is the group $I_2(m)$ of \cite[Ex.~1.2.7]{BB}. There, every
  element is represented by an alternating word of length $\le m$, the two alternating words of
  length $m$ represent the same element, and the group has order $2m$. The $2m-1$ alternating words
  of length $< m$, together with $\altw stm$, therefore represent the $2m$ elements bijectively.

  Let $k \le m$ and suppose $\altw stk$ is not reduced in $(W_J,J)$. A reduced expression of the same
  element contains no factor $rr$, since deleting it would shorten it, so it is an alternating word,
  of length $< k \le m$. It is a different word from $\altw stk$ on the list above, but represents
  the same element: a contradiction. So $\altw stk$ is reduced in $(W_J,J)$, hence in $(W,S)$, as
  the length functions agree. The last sentence follows from this with $k = j+1$ and
  \Cref{prop:min}(2) for $X = \{x\}$.
\end{proof}

\section{The category \texorpdfstring{$\Cx(W,S)$}{C(W,S)} and its presheaves}

\begin{definition}\label{def:C}
  $\Cx = \Cx(W,S)$ has objects $\Sf$, morphisms $\Cx(J',J) := W_J \cap W^{J'}$ if $J' \subseteq J$
  and $\emptyset$ otherwise, and composition the product of $W$.
\end{definition}

\begin{lemma}\label{lem:C}
  $\Cx$ is a category. Moreover $\Cx(\emptyset,J) = W_J$ and $\Cx(J,J) = \{1\}$.
\end{lemma}
\begin{proof}
  $1 \in W_J \cap W^J$. Let $v \in \Cx(J',J)$ and $w \in \Cx(J,J'')$; then $wv \in W_{J''}$. For $u
  \in W_{J'}$, \Cref{prop:min}(2) gives $\ell(wvu) = \ell(w) + \ell(vu) = \ell(w) + \ell(v) +
  \ell(u)$, using $vu \in W_J$ and then $u \in W_{J'}$, and likewise $\ell(wv) = \ell(w) + \ell(v)$.
  So $\ell(wvu) = \ell(wv) + \ell(u)$ for all $u \in W_{J'}$, and $wv \in W^{J'}$ by
  \Cref{prop:min}(2). Every coset of $W_\emptyset = \{1\}$ is a singleton, so $W^\emptyset = W$.
  An element of $W_J \cap W^J$ has minimal length in $W_J$, so it is $1$.
\end{proof}

The verification is the computation in the proof of \cite[Lem.~3.2]{Paris}.

Fix a presheaf $P$ on $\Cx$. For $v \in \Cx(J',J)$ and $d \in P(J)$ write $d \cdot v \in P(J')$, so
$(d\cdot v)\cdot u = d \cdot (vu)$ and $d \cdot 1 = d$. The elements of $P(\emptyset)$ are the
\textbf{runs}. Every $v \in \Cx(J',J)$ also lies in $\Cx(\emptyset,J) = W_J$, and $(d\cdot v)\cdot 1
= d\cdot v$ as runs; we write $d\cdot v$ in both cases. The \textbf{bottom run} of $d$ is $d\cdot1
\in P(\emptyset)$.

The category of elements $\int P$ has objects the $d \in P(J)$, $J \in \Sf$, and a morphism $c \to
d$, for $c \in P(J')$ and $d \in P(J)$, is a $v \in \Cx(J',J)$ with $d\cdot v = c$; composition is
the product. Let $W_P$ be the class of morphisms $v = 1$ of $\int P$, and
\[
  \Br(P) \;:=\; (\textstyle\int P)\op\bigl[(W_P\op)\inv\bigr],
\]
with localization functor $Q$, contravariant on $\int P$ as in \texttt{mainv2}. For $d \in P(J)$
and $w \in W_J$, write $\rho^d_w : d\cdot w \to d$ for the morphism $w$ of $\int P$. Then
$\rho^d_1 \in W_P$.

Fix a total order on $S$.

\begin{definition}\label{def:art}
  The \textbf{Artin polygraph} $\Art(P)$ has the runs as $0$-generators. A $1$-generator from $X$ to
  $Y$ is an $\alpha \in P(\{s\})$, $s \in S$, with $\alpha\cdot1 = X$ and $\alpha\cdot s = Y$. A
  $2$-generator is an $\varepsilon \in P(\{s,t\})$ with $s < t$ and $m := m_{s,t} < \infty$, and
  \[
    \src(\varepsilon) = \bigl(\varepsilon\cdot\altw st0,\ \dots,\ \varepsilon\cdot\altw st{m-1}\bigr),
    \qquad
    \tgt(\varepsilon) = \bigl(\varepsilon\cdot\altw ts0,\ \dots,\ \varepsilon\cdot\altw ts{m-1}\bigr).
  \]
\end{definition}

By \Cref{lem:dihedral}, $\altw stj \in \Cx(\{x\},\{s,t\})$ for the $(j+1)$-st letter $x$ of $\altw
stm$, so $\varepsilon\cdot\altw stj$ is a $1$-generator from $\varepsilon\cdot\altw stj$ to
$\varepsilon\cdot\altw st{j+1}$, and $\src(\varepsilon)$ is a path from $\varepsilon\cdot 1$ to
$\varepsilon\cdot\altw stm$. Likewise for $\tgt(\varepsilon)$, which ends at the same run since $\altw
stm = \altw tsm$ in $W$.

\section{Cocycles and Artin families}

Let $D$ be a category.

\begin{definition}\label{def:cocycle}
  A \textbf{$P$-cocycle} in $D$ is an object $x_X$ of $D$ for each run $X$ and an arrow
  $\sigma_d(w) : x_{d\cdot1} \to x_{d\cdot w}$ for each $J \in \Sf$, $d \in P(J)$ and $w \in W_J$,
  such that
  \begin{enumerate}[label=(C\arabic*)]
    \item $\sigma_d(1) = \id$;
    \item $\sigma_d(vu) = \sigma_{d\cdot v}(u) \circ \sigma_d(v)$ for all $J' \subseteq J$,
      $v \in \Cx(J',J)$ and $u \in W_{J'}$.
  \end{enumerate}
\end{definition}

\begin{definition}\label{def:family}
  A \textbf{$P$-Artin family} in $D$ is an object $x_X$ of $D$ for each run $X$ and an arrow $a_\alpha :
  x_{\alpha\cdot1} \to x_{\alpha\cdot s}$ for each $\alpha \in P(\{s\})$, $s \in S$, such that for
  every $2$-generator $\varepsilon \in P(\{s,t\})$ of $\Art(P)$, with $m = m_{s,t}$,
  \[
    a_{\varepsilon\cdot\altw st{m-1}} \circ \cdots \circ a_{\varepsilon\cdot\altw st0}
    \;=\;
    a_{\varepsilon\cdot\altw ts{m-1}} \circ \cdots \circ a_{\varepsilon\cdot\altw ts0} .
  \]
\end{definition}

Write $\Coc_P(D)$ and $\Fam_P(D)$ for the sets of these. A functor $F : D \to D'$ acts on both by
applying $F$ to every object and arrow.

Write $q : \Art(P)_1^* \to \pres{\Art(P)}$ for the quotient of the free category on the
$1$-generators.

\begin{proposition}\label{prop:pres}
  Sending a functor $F : \pres{\Art(P)} \to D$ to $(F(X), F(\alpha))$ is a bijection onto
  $\Fam_P(D)$.
\end{proposition}
\begin{proof}
  The arrows $F(\alpha)$ satisfy the family relations because $q(\src\varepsilon) =
  q(\tgt\varepsilon)$. Conversely a family extends uniquely to the free category
  \cite[Lem.~2.1.3]{Polygraphs}, and then uniquely to $\pres{\Art(P)}$ \cite[Lem.~2.3.4]{Polygraphs},
  since its relations say exactly that the source and target of each $2$-generator have equal image.
\end{proof}

\begin{theorem}[Matsumoto with coefficients]\label{thm:matsumoto}
  Sending $(x,\sigma)$ to $(x,a)$ with $a_\alpha := \sigma_\alpha(s)$, for $\alpha \in P(\{s\})$, is a
  bijection $\Coc_P(D) \to \Fam_P(D)$, compatible with the action of functors $D \to D'$.
\end{theorem}
\begin{proof}
  \emph{Well defined.} $\sigma_\alpha(s) : x_{\alpha\cdot1} \to x_{\alpha\cdot s}$ has the right type.
  Let $\varepsilon \in P(\{s,t\})$ be a $2$-generator and $x_{j+1}$ the $(j+1)$-st letter of $\altw
  stm$. By \Cref{lem:dihedral}, $\altw stj \in \Cx(\{x_{j+1}\},\{s,t\})$, so (C2) with $v = \altw
  stj$ and $u = x_{j+1}$ gives
  \[
    \sigma_\varepsilon\bigl(\altw st{j+1}\bigr)
      = a_{\varepsilon\cdot\altw stj} \circ \sigma_\varepsilon\bigl(\altw stj\bigr) .
  \]
  By induction and (C1), the left side of the family relation is $\sigma_\varepsilon(\altw stm)$.
  Likewise the right side is $\sigma_\varepsilon(\altw tsm)$, and $\altw stm = \altw tsm$ in $W$.

  \emph{The inverse.} Let $(x,a) \in \Fam_P(D)$, $d \in P(J)$ and $w \in W_J$. For a reduced
  expression $\mu = x_1\cdots x_k$ of $w$, put $w_i := x_1\cdots x_i$ and
  \[
    \sigma_d(\mu) := a_{d\cdot w_{k-1}} \circ \cdots \circ a_{d\cdot w_0} .
  \]
  This is well typed: $x_i \in J$ by \Cref{prop:min}(4); $w_{i-1}x_i = w_i$ is reduced by
  \Cref{lem:prefix}, so $\ell(w_{i-1}x_i) > \ell(w_{i-1})$ and $w_{i-1} \in W^{\{x_i\}}$ by
  \Cref{prop:min}(2); so $w_{i-1} \in \Cx(\{x_i\},J)$, and $d\cdot w_{i-1}$ is a $1$-generator from
  $d\cdot w_{i-1}$ to $d\cdot w_i$.

  It does not depend on $\mu$. By \Cref{thm:tits} it suffices to compare $\mu = \nu_1\altw
  stm\nu_2$ with $\mu' = \nu_1\altw tsm\nu_2$, both reduced, and we may take $s < t$. Let $v$ be the
  element represented by $\nu_1$. Then $v \in W_J$, as the letters of $\mu$ lie in $J$
  (\Cref{prop:min}(4)), and $vs$, $vt$ are represented by prefixes of
  $\mu$, $\mu'$, so $\ell(vs) = \ell(vt) = \ell(v)+1$ by \Cref{lem:prefix}; hence $v \in
  W^{\{s,t\}}$ by \Cref{prop:min}(2), and $\varepsilon := d\cdot v \in P(\{s,t\})$ is a
  $2$-generator. The two composites have the same factors along $\nu_1$, and along $\nu_2$ since
  $\nu_1\altw stm$ and $\nu_1\altw tsm$ represent the same element. Along the middle, the factors are
  $a_{d\cdot v\altw stj} = a_{\varepsilon\cdot\altw stj}$ and $a_{\varepsilon\cdot\altw tsj}$, whose
  composites agree by the family relation of $\varepsilon$. Write $\sigma_d(w)$ for the common
  value.

  (C1) holds as the empty product. For (C2), let $v \in \Cx(J',J)$ and $u \in W_{J'}$. By
  \Cref{prop:min}(2), $\ell(vu) = \ell(v) + \ell(u)$, so reduced expressions of $v$ and $u$
  concatenate to a reduced expression of $vu$. Reading $\sigma_d(vu)$ along it, the factors past $v$
  are $a_{d\cdot vu_i} = a_{(d\cdot v)\cdot u_i}$, which compose to $\sigma_{d\cdot v}(u)$.

  \emph{Mutually inverse.} Starting from a family, $\sigma_\alpha(s)$ is read along the expression
  $s$, so it is $a_{\alpha\cdot1} = a_\alpha$. Starting from a cocycle, (C2) with $v = w_{i-1}$ and
  $u = x_i$ gives $\sigma_d(w_i) = \sigma_{d\cdot w_{i-1}}(x_i) \circ \sigma_d(w_{i-1})$, so by
  induction $\sigma_d(w)$ is the composite of the $\sigma_{d\cdot w_{i-1}}(x_i) = a_{d\cdot
  w_{i-1}}$.

  Compatibility with functors is clear from the formulas.
\end{proof}

At $P = 1$ and $D$ the Artin monoid $A^+$, viewed as a one-object category composed left to right
as in \texttt{mainv2}, with $a_s := \sigma_s$, the inverse construction is the set-section $\tau : W
\to A^+$ of \cite[\S2.2]{Paris}; (C2) then says $\tau(vu) = \tau(v)\tau(u)$
for $v \in W^{J'}$ and $u \in W_{J'}$.

\section{The presentation}

\begin{lemma}\label{lem:s}
  Putting $x_X := Q(X)$ and $s_d(w) := Q(\rho^d_w) \circ Q(\rho^d_1)\inv$ defines a $P$-cocycle $s$ in
  $\Br(P)$.
\end{lemma}
\begin{proof}
  (C1) is clear. For (C2), let $v \in \Cx(J',J)$, $u \in W_{J'}$ and $c := d\cdot v$, and let $g :
  c \to d$ be the morphism $v$. In $\int P$, $\rho^d_{vu} = g \circ \rho^c_u$ and $\rho^d_v = g
  \circ \rho^c_1$, as both sides are the same element with the same target. Since $Q$ reverses
  composition,
  \[
    s_d(vu) = Q(\rho^c_u)\,Q(g)\,Q(\rho^d_1)\inv
      = \bigl[Q(\rho^c_u)\,Q(\rho^c_1)\inv\bigr]\bigl[Q(\rho^c_1)\,Q(g)\,Q(\rho^d_1)\inv\bigr]
      = s_c(u) \circ s_d(v) . \qedhere
  \]
\end{proof}

\begin{theorem}\label{thm:A}
  Let $E : \pres{\Art(P)} \to \Br(P)$ be the functor that \Cref{prop:pres} assigns to the
  family of the cocycle $s$ (\Cref{thm:matsumoto}); so $E(X) = Q(X)$ and $E(\alpha) =
  Q(\rho^\alpha_s)\,Q(\rho^\alpha_1)\inv$. Then $E$ is an equivalence of categories. For a map $\varphi
  : P \to P'$ of presheaves, $E_{P'} \circ \pres{\Art(\varphi)} = \Br(\varphi) \circ E_P$.
\end{theorem}
\begin{proof}
  Let $\Gamma$ be the cocycle in $\pres{\Art(P)}$ whose family is $(X, q(\alpha))$
  (\Cref{thm:matsumoto}). Define $\Theta : (\int P)\op \to \pres{\Art(P)}$ by $\Theta(c) := c\cdot1$
  and $\Theta(g) := \Gamma_d(v)$ for $g = v : c \to d$; the types match because $c\cdot 1 = d\cdot
  v$. It is a functor by (C2), since composition in $\int P$ is the product, and it sends $W_P$ to
  identities by (C1). By \cite[\S I.1.1, Lem.~I.1.2]{GZ} there is a unique $\Phi : \Br(P) \to
  \pres{\Art(P)}$ with $\Phi Q = \Theta$.

  \emph{$\Phi E = \id$.} $\Phi E(X) = X\cdot1 = X$, and $\Phi E(\alpha) = \Gamma_\alpha(s) \circ
  \Gamma_\alpha(1)\inv = q(\alpha)$. So $\Phi E$ and $\id$ agree on generators, and they are equal
  by the uniqueness in \cite[Lem.~2.1.3, Lem.~2.3.4]{Polygraphs}.

  \emph{$E\Phi \cong \id$.} $E\circ\Gamma$ is a cocycle in $\Br(P)$ with the same family as $s$, so
  $E\circ\Gamma = s$ by \Cref{thm:matsumoto}. Hence for $g = v : c \to d$, using $\rho^d_v = g
  \circ \rho^c_1$,
  \[
    E\Theta(g) = s_d(v) = Q(\rho^c_1)\,Q(g)\,Q(\rho^d_1)\inv .
  \]
  So the arrows $\eta_c := Q(\rho^c_1) : Q(c) \to E\Theta(c)$ form a natural isomorphism $Q \cong
  E\Theta = E\Phi Q$. Since precomposition with $Q$ is an isomorphism onto a full subcategory
  \cite[Lem.~I.1.2]{GZ}, $\eta = \eta' Q$ for a unique $\eta' : \id \Rightarrow E\Phi$, invertible
  because $\eta\inv$ descends too.

  \emph{Naturality.} $\int\varphi$ carries $W_P$ into $W_{P'}$ and $\rho^d_w$ to
  $\rho^{\varphi d}_w$, and $\Art(\varphi)$, defined by $\varphi$ on generators, is a morphism of
  polygraphs because $\varphi$ commutes with restriction. Both composites send $\alpha$ to
  $Q(\rho^{\varphi\alpha}_s)\,Q(\rho^{\varphi\alpha}_1)\inv$, so they agree on generators.
\end{proof}

\begin{corollary}\label{cor:artin}
  At the terminal presheaf $P = 1$, $\Art(1)$ has one $0$-generator, the $1$-generators $S$, and one
  $2$-generator $\altw stm = \altw tsm$ for each $s < t$ with $m = m_{s,t} < \infty$. So
  $\pres{\Art(1)}$ is the Artin monoid $A^+$ of \cite[\S2.4]{Paris}, and $\Cx\op[1\inv] \simeq A^+$.
\end{corollary}
\begin{proof}
  Each $P(J)$ is a point, so $\varepsilon\cdot\altw stj$ is the unique element of $P(\{x_{j+1}\})$,
  that is, the generator $x_{j+1}$. Then apply \Cref{thm:A}.
\end{proof}

\begin{remark}
  For the torsor $\tw(J) := W$, with $u\cdot w := uw$, $\int\tw$ is the poset $(W\times\Sf,\preceq)$
  of \cite[Lem.~3.2]{Paris}: a morphism $(u,X) \to (v,Y)$ is a $w \in W_Y \cap W^X$ with $vw = u$,
  that is $w = v\inv u$. Its Artin polygraph has the edges $a(u,s) : u \to us$ and the $2$-cells
  $B(u,\{s,t\})$ of the $2$-skeleton of the Salvetti complex described in \cite[\S3.3]{Paris}.
\end{remark}

\section{Application to precubical sets}\label{sec:cubes}

Refinements preserve the number of events, so $\Ch(K)$, $\Br(K)$ and $\Art(K)$ split as disjoint
unions over $N$, and we work at one $N$ at a time. For a bipointed precubical set $K$, let $N(K)$ be the presheaf $a \mapsto \BPSet(\bigvee a, K)$ on

\int N(K)$; the order-preserving refinements are the morphisms over $1$, so $\Br(K) = \Br(N(K))$;
and $\Art(K) = \Art(N(K))$, with $\src$ and $\tgt$ given by \Cref{def:art}. Theorem~1.1 of
\texttt{mainv2} follows: (i) is \Cref{thm:A} at $P = N(K)$; (ii) is its naturality clause, $N$
being a functor; (iii) is \Cref{cor:artin}, since $N(Z) = 1$.
