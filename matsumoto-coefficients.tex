\section{Coxeter background}

$s_t := (t,\,t+1) \in S_N$, $S := \{s_1 < \cdots < s_{N-1}\}$. $(S_N,S)$ is a Coxeter system
\cite[Prop.~1.5.4]{BB} with $m_{s_i,s_k} = \cox(i,k)$, equal to $3$ if $|i-k| = 1$ and $2$ otherwise
\cite[(1.29)]{BB}. $W_X := \langle X\rangle$. $w$ is \textbf{$(\emptyset,X)$-minimal} if it has minimal
length in $wW_X$ \cite[\S2.2]{Paris}; $W^X$ is the set of these. $\altw stk$ is the alternating word
$sts\cdots$ of length $k \ge 0$ (\cite[\S1]{Paris} for $k \ge 2$), and the element it represents. An
\textbf{elementary $M$-transformation} replaces $\nu_1\altw st{m_{s,t}}\nu_2$ by
$\nu_1\altw ts{m_{s,t}}\nu_2$ \cite[\S2.2]{Paris}.

\begin{theorem}[{\cite[Thm.~2.7]{Paris}}]\label{thm:tits}
  Let $w \in W$, and let $\mu, \mu'$ be two reduced expressions of $w$. Then there is a finite
  sequence of elementary $M$-transformations joining $\mu$ to $\mu'$.
\end{theorem}

\begin{proposition}[{\cite[Prop.~2.9]{Paris}}]\label{prop:min}
  Let $(W,S)$ be a Coxeter system.
  \begin{enumerate}
    \item[(2)] Let $X \subset S$, and let $w \in W$. Then $w$ is $(\emptyset,X)$-minimal if and
      only if $\ell(ws) > \ell(w)$ for all $s \in X$, and $\ell(ws) > \ell(w)$ for all $s \in X$ if
      and only if $\ell(wu) = \ell(w) + \ell(u)$ for all $u \in W_X$.
    \item[(4)] Let $X \subset S$, and let $w \in W_X$. If $\mu = s_1 \cdots s_\ell$ is a reduced
      expression of $w$, then $s_1, \dots, s_\ell \in X$.
  \end{enumerate}
\end{proposition}

\begin{proposition}\label{prop:typeA}
  \begin{enumerate}
    \item \cite[Prop.~1.4.2(ii)]{BB} $\ell(uw) \equiv \ell(u) + \ell(w) \pmod 2$.
    \item \cite[Prop.~1.5.2]{BB} $\ell(x) = \mathrm{inv}(x) := \#\{(i,j) : i < j,\ x(i) > x(j)\}$.
    \item \cite[Prop.~1.5.3]{BB} $D_R(x) = \{s_i : x(i) > x(i+1)\}$, where $D_R(x) := \{s \in S :
      \ell(xs) < \ell(x)\}$.
  \end{enumerate}
\end{proposition}

\begin{lemma}\label{lem:ascent}
  $\ell(xs_t) > \ell(x) \iff x(t) < x(t+1)$.
\end{lemma}
\begin{proof}
  \Cref{prop:typeA}(1) gives $\ell(xs_t) \neq \ell(x)$; then \Cref{prop:typeA}(3).
\end{proof}

\begin{lemma}\label{lem:prefix}
  A prefix of a reduced expression is reduced.
\end{lemma}
\begin{proof}
  A shorter expression for the prefix shortens the whole.
\end{proof}

\begin{lemma}\label{lem:dihedral}
  $\altw stk$ is reduced for $k \le m := m_{s,t}$. Hence $\altw stj \in W^{\{x\}}$ for $j < m$, $x$
  the $(j+1)$-st letter of $\altw stm$.
\end{lemma}
\begin{proof}
  By \Cref{lem:prefix,prop:typeA}(2), check $\mathrm{inv}(\altw stm) = m$: for $m = 2$ it is two
  disjoint adjacent transpositions; for $m = 3$ it is $(i,\,i+2)$. Then \Cref{prop:min}(2).
\end{proof}

\section{\texorpdfstring{$\EN$}{E\_N} and its presheaves}

\begin{definition}[{\cite[Def.~6.11]{PZ}}]\label{def:EN}
  Let $\EN$ be the category whose objects are subsets of $\{1,\dots,N-1\}$. For $B = \{b_1 < \cdots <
  b_{r-1}\} \supseteq B' = \{b'_1 < \cdots < b'_{r'-1}\}$, morphisms from $B$ to $B'$ are permutations
  $\phi$ of $\{1,\dots,N\}$ satisfying
  \begin{enumerate}
    \item[(x)] $\phi(\{b'_{p-1}+1,\dots,b'_p\}) = \{b'_{p-1}+1,\dots,b'_p\}$ for $p \in \{1,\dots,r'\}$
      (with $b'_0 = 0$, $b'_{r'} = N$);
    \item[(y)] $\phi(b_{p-1}+1) < \phi(b_{p-1}+2) < \cdots < \phi(b_p)$ for $p \in \{1,\dots,r\}$ (with
      $b_0 = 0$, $b_r = N$).
  \end{enumerate}
  For $B \not\supseteq B'$, $\EN(B,B') = \emptyset$. Composition is composition of permutations.
\end{definition}

By \cite[Def.~6.11]{PZ}, (x) $\iff$ (x$'$): $i \le b' < j$ for some $b' \in B'$ implies $\phi(i) <
\phi(j)$. Put $B(J) := \{t : s_t \notin J\}$.

\begin{lemma}\label{lem:EN}
  For $J' \subseteq J$, $\EN(B(J'),B(J)) = W_J \cap W^{J'}$.
\end{lemma}
\begin{proof}
  (y) for $B(J')$ says $\phi(t) < \phi(t+1)$ for $s_t \in J'$; use \Cref{lem:ascent,prop:min}(2).
  (x) for $B(J)$: each $s_t \in J$ satisfies (x), and (x) is closed under products, so $W_J$ satisfies
  (x). Conversely, induct on $\ell(\phi)$: if $\phi \neq 1$,
  pick $s_t \in D_R(\phi)$; (x$'$) forces $s_t \in J$; $\phi s_t$ satisfies (x) and is shorter.
\end{proof}

Henceforth $J$ denotes $B(J)$, so $\EN(J',J) = W_J \cap W^{J'}$ ($\emptyset$ unless $J' \subseteq J$),
$\EN(\emptyset,J) = W_J$, composition is the product.

Fix a presheaf $P$ on $\EN$; write $d\cdot v$ for restriction. \textbf{Runs}: $P(\emptyset)$. For $v
\in \EN(J',J) \subseteq W_J = \EN(\emptyset,J)$, $(d\cdot v)\cdot 1 = d\cdot v$; both are written
$d\cdot v$. $\int P$: objects $d \in P(J)$; morphisms $v : d\cdot v \to d$, $v \in \EN(J',J)$;
composition the product. $W_P := \{v = 1\}$. $\Br(P) := (\int P)\op[(W_P\op)\inv]$, $Q$ contravariant.
$\rho^d_w : d\cdot w \to d$ is the morphism $w \in W_J$; $\rho^d_1 \in W_P$.

\begin{definition}\label{def:art}
  $\Art(P)$: $0$-generators the runs; $1$-generators $\alpha \in P(\{s\})$, $\alpha\cdot1 \to
  \alpha\cdot s$; $2$-generators $\varepsilon \in P(\{s,t\})$, $s < t$, $m := m_{s,t}$, with
  \[
    \src(\varepsilon) = \bigl(\varepsilon\cdot\altw stj\bigr)_{j<m}, \qquad
    \tgt(\varepsilon) = \bigl(\varepsilon\cdot\altw tsj\bigr)_{j<m}.
  \]
  These are paths from $\varepsilon\cdot 1$ to $\varepsilon\cdot\altw stm = \varepsilon\cdot\altw tsm$
  by \Cref{lem:dihedral} and \cite[(1.29)]{BB}. $q : \Art(P)_1^* \to \pres{\Art(P)}$ is the quotient.
\end{definition}

\section{Cocycles and Artin families}

\begin{definition}\label{def:cocycle}
  A \textbf{$P$-cocycle} in $D$: objects $x_X$ ($X$ a run) and arrows $\sigma_d(w) : x_{d\cdot1} \to
  x_{d\cdot w}$ ($d \in P(J)$, $w \in W_J$) with
  \begin{enumerate}[label=(C\arabic*)]
    \item $\sigma_d(1) = \id$;
    \item $\sigma_d(vu) = \sigma_{d\cdot v}(u) \circ \sigma_d(v)$ for $v \in \EN(J',J)$, $u \in
      W_{J'}$.
  \end{enumerate}
  A \textbf{$P$-Artin family} in $D$: objects $x_X$ and arrows $a_\alpha : x_{\alpha\cdot1} \to
  x_{\alpha\cdot s}$ ($\alpha \in P(\{s\})$) with, for each $2$-generator $\varepsilon$,
  \[
    a_{\varepsilon\cdot\altw st{m-1}} \circ \cdots \circ a_{\varepsilon\cdot\altw st0}
    \;=\;
    a_{\varepsilon\cdot\altw ts{m-1}} \circ \cdots \circ a_{\varepsilon\cdot\altw ts0} .
  \]
  $\Coc_P(D)$, $\Fam_P(D)$ denote the sets of these; functors act on both.
\end{definition}

\begin{proposition}\label{prop:pres}
  $F \mapsto (F(X), F(q\alpha))$ is a bijection from functors $\pres{\Art(P)} \to D$ to $\Fam_P(D)$.
\end{proposition}
\begin{proof}
  \cite[Lem.~2.1.3, Lem.~2.3.4]{Polygraphs}.
\end{proof}

\begin{theorem}[reduced words in $P$]\label{thm:matsumoto}
  $(x,\sigma) \mapsto (x,a)$, $a_\alpha := \sigma_\alpha(s)$, is a bijection $\Coc_P(D) \to
  \Fam_P(D)$, compatible with functors.
\end{theorem}
\begin{proof}
  \emph{Well defined.} With $x_{j+1}$ the $(j+1)$-st letter of $\altw stm$, (C2) for $v = \altw stj$
  (\Cref{lem:dihedral}), $u = x_{j+1}$ gives $\sigma_\varepsilon(\altw st{j+1}) =
  a_{\varepsilon\cdot\altw stj} \circ \sigma_\varepsilon(\altw stj)$. So both sides of the family
  relation equal $\sigma_\varepsilon(\altw stm) = \sigma_\varepsilon(\altw tsm)$.

  \emph{Inverse.} For a reduced expression $x_1\cdots x_k$ of $w \in W_J$, $w_i := x_1\cdots x_i$, set
  $\sigma_d(w) := a_{d\cdot w_{k-1}} \circ \cdots \circ a_{d\cdot w_0}$. Typing: $x_i \in J$
  (\Cref{prop:min}(4)); $w_{i-1} \in W^{\{x_i\}}$ (\Cref{lem:prefix,prop:min}(2)), so $w_{i-1} \in
  \EN(\{x_i\},J)$. Independence: by \Cref{thm:tits}, compare $\nu_1\altw stm\nu_2$ and $\nu_1\altw
  tsm\nu_2$; $v := \nu_1$ lies in $W_J$ (\Cref{prop:min}(4)) and in $W^{\{s,t\}}$ ($vs$, $vt$ are
  reduced prefixes; \Cref{lem:prefix,prop:min}(2)); the middle factors are $a_{\varepsilon\cdot\altw
  stj}$ and $a_{\varepsilon\cdot\altw tsj}$ for $\varepsilon := d\cdot v$, equal by the family
  relation. (C1): empty product. (C2): $\ell(vu) = \ell(v) + \ell(u)$ (\Cref{prop:min}(2)), so
  concatenate reduced expressions; $a_{d\cdot vu_i} = a_{(d\cdot v)\cdot u_i}$.

  \emph{Mutually inverse.} $\sigma_\alpha(s) = a_\alpha$. Conversely (C2) with $v = w_{i-1}$, $u = x_i$
  gives $\sigma_d(w_i) = a_{d\cdot w_{i-1}} \circ \sigma_d(w_{i-1})$.
\end{proof}

\section{The presentation}

\begin{lemma}\label{lem:s}
  $x_X := Q(X)$, $s_d(w) := Q(\rho^d_w)\,Q(\rho^d_1)\inv$ is a $P$-cocycle in $\Br(P)$.
\end{lemma}
\begin{proof}
  (C2): with $c := d\cdot v$ and $g := v : c \to d$, $\rho^d_{vu} = g\rho^c_u$ and $\rho^d_v =
  g\rho^c_1$, so $s_d(vu) = [Q(\rho^c_u)Q(\rho^c_1)\inv][Q(\rho^c_1)Q(g)Q(\rho^d_1)\inv] = s_c(u)
  \circ s_d(v)$.
\end{proof}

\begin{theorem}\label{thm:A}
  Let $E : \pres{\Art(P)} \to \Br(P)$ correspond to $s$ (\Cref{prop:pres,thm:matsumoto}). Then $E$ is
  an equivalence, and $E_{P'}\pres{\Art(\varphi)} = \Br(\varphi)E_P$ for $\varphi : P \to P'$.
\end{theorem}
\begin{proof}
  Let $\Gamma$ be the cocycle in $\pres{\Art(P)}$ with family $(X, q\alpha)$. $\Theta(c) := c\cdot1$,
  $\Theta(v : c \to d) := \Gamma_d(v)$ is a functor $(\int P)\op \to \pres{\Art(P)}$ by (C2),
  killing $W_P$ by (C1); let $\Phi Q = \Theta$ \cite[\S I.1.1, Lem.~I.1.2]{GZ}.

  $\Phi E = \id$: $\Phi E(q\alpha) = \Gamma_\alpha(s) = q\alpha$; uniqueness in
  \cite[Lem.~2.1.3, Lem.~2.3.4]{Polygraphs}.

  $E\Phi \cong \id$: $E\Gamma$ and $s$ have the same family, so $E\Gamma = s$
  (\Cref{thm:matsumoto}). With $\rho^d_v = g\rho^c_1$, $E\Theta(g) = Q(\rho^c_1)Q(g)Q(\rho^d_1)\inv$,
  so $\eta_c := Q(\rho^c_1)$ is a natural isomorphism $Q \cong E\Phi Q$; descend by
  \cite[Lem.~I.1.2]{GZ}.

  Naturality: check on generators; $\int\varphi$ preserves $W$ and $\rho$.
\end{proof}

\begin{corollary}\label{cor:artin}
  $\Art(1)$ is the component at $N$ of \texttt{def:artin-poly}.
\end{corollary}
\begin{proof}
  Each $P(J)$ is a point; $\varepsilon\cdot\altw stj$ is the generator $x_{j+1}$.
\end{proof}

\section{Precubical sets}\label{sec:cubes}

\begin{proposition}\label{prop:z2}
  \begin{enumerate}
    \item \cite[\S2.4]{Z2} $\cube{n}[k]$ is the set of functions $\{1,\dots,n\} \to \{0,1,*\}$ that take value $*$
      for exactly $k$ arguments; the face map $d^\varepsilon_i$ converts the $i$-th occurrence of
      $*$ into $\varepsilon$; $u_n$ is the only element of $\cube{n}[n]$; for $c \in K[n]$ there
      is a unique map $f_c : \cube{n} \to K$ with $f_c(u_n) = c$; $d^\varepsilon :=
      d^\varepsilon_{\{1,\dots,n\}}$.
    \item \cite[Def.~7.1, Def.~7.2]{Z2} $(K \vee L)[n] = K[n] \amalg L[n]$ for $n > 0$, $(K \vee L)[0] = (K[0] \amalg L[0])/1_K
      \sim 0_L$, $0_{K\vee L} = 0_K$, $1_{K \vee L} = 1_L$; $\cube{n}$ is bipointed by $0 =
      (0,\dots,0)$, $1 = (1,\dots,1)$.
    \item \cite[\S7, after Def.~7.3]{Z2} Cube chains $c : \cube{n_1} \vee \cdots \vee \cube{n_l} \to K$ correspond $1$-$1$ to
      sequences of cubes $(c_1,\dots,c_l)$ with $d^0(c_1) = 0_K$, $d^1(c_l) = 1_K$ and $d^1(c_i) =
      d^0(c_{i+1})$, via $c_i = c(u_{n_i})$.
  \end{enumerate}
\end{proposition}

So $d^0(x)$, $d^1(x)$ replace every $*$ of $x$ by $0$, $1$, and the face embedding $f_x :
\cube{k} \to \cube{n}$ leaves free the positions of $*$ in $x$; hence $\Ev(f_x)$ is the monotone
injection onto them (\S2 of \texttt{mainv2}).

Fix shapes $a = (a_1,\dots,a_r)$, $b = (b_1,\dots,b_q)$ on $N$ events, with blocks $A_i$, $B_j$
(intervals of $[N]$, in order). A cube of $\bigvee b$ of dimension $a_i > 0$ is an $x \in
\cube{b_j}[a_i]$ for one $j$; write $X(x), O(x) \subseteq B_j$ for the positions of $*$ and of
$1$, shifted into $B_j$.

\begin{lemma}\label{lem:chains}
  Cubes $x_i \in \cube{b_{j(i)}}[a_i]$, $i \le r$, form a cube chain in $\bigvee b$ if and only if
  $j$ is non-decreasing and, for each $j$, the $X(x_i)$ with $j(i) = j$ partition $B_j$ and $O(x_i)
  = \bigcup\{X(x_{i'}) : i' < i,\ j(i') = j(i)\}$.
\end{lemma}
\begin{proof}
  $d^0(x_i)$, $d^1(x_i)$ are the vertices of $\cube{b_{j(i)}}$ with $1$s at $O(x_i)$, at
  $O(x_i) \sqcup X(x_i)$. By \Cref{prop:z2}(2), a vertex of $\cube{b_j}$ is identified with a
  vertex of another block only if it is $1$ of $\cube{b_j}$ and $0$ of $\cube{b_{j+1}}$, or
  conversely. ($\Leftarrow$): $d^0(x_1) = 0$, $d^1(x_r) = 1$, and
  $d^1(x_i) = d^0(x_{i+1})$, either inside one block or as $1$ of $\cube{b_j}$ $=$ $0$ of
  $\cube{b_{j+1}}$; apply \Cref{prop:z2}(3). ($\Rightarrow$):
  $d^0(x_1) = 0$ forces $j(1) = 1$, $O(x_1) = \emptyset$. Given $x_i$, if $O(x_i) \sqcup X(x_i) \neq
  B_{j(i)}$ then $d^0(x_{i+1})$ lies only in block $j(i)$, so $j(i+1) = j(i)$ and $O(x_{i+1}) =
  O(x_i) \sqcup X(x_i)$. If $O(x_i) \sqcup X(x_i) = B_{j(i)}$, then $j(i+1) = j(i)$ is impossible,
  as $X(x_{i+1}) \neq \emptyset$ is disjoint from $O(x_{i+1})$; so $j(i+1) = j(i)+1$ and
  $O(x_{i+1}) = \emptyset$, or $i = r$ and $j(i) = q$ by $d^1(x_r) = 1$. So within block $j$ the $O(x_i)$ grow from
  $\emptyset$ to $B_j$ by the disjoint $X(x_i)$.
\end{proof}

\begin{proposition}\label{prop:cubes}
  Let $B_a := \{t \in [N-1] : \blk_a(t) \neq \blk_a(t+1)\}$. Then $a \mapsto B_a$, $f \mapsto f_*$ is
  an isomorphism from the full subcategory of $\Ch(Z)$ on shapes on $N$ events onto $\EN$.
\end{proposition}
\begin{proof}
  Objects: \texttt{lem:cuts}. Functoriality: $\Ev$. Let $f : \bigvee a \to \bigvee b$ have cubes
  $x_i = f(u_{a_i})$ (\Cref{prop:z2}(3)). Then $f_*$ is $\Ev(f_{x_i})$ on the summand $[a_i]$, so it
  maps $A_i$ increasingly onto $X(x_i)$. By \Cref{lem:chains}, the $A_i$ with $j(i) = j$ are
  consecutive with total size $b_j$, so their union is $B_j$: hence $B_a \supseteq B_b$, (x) and
  (y). Injective: $f$ is determined by the $x_i$ (\Cref{prop:z2}(3)), and $x_i$ by $j(i)$, $X(x_i) =
  f_*(A_i)$ and $O(x_i)$, which \Cref{lem:chains} determines from the $X$s. Surjective: given $\phi
  \in \EN(B_a,B_b)$, let $j(i)$ be the block of $b$ containing $A_i$, and $x_i$ the cube with $X(x_i) =
  \phi(A_i) \subseteq B_{j(i)}$ (by (x)) and $O(x_i)$ as in \Cref{lem:chains}; its hypotheses hold as
  $\phi$ is a bijection preserving each $B_j$. The resulting $f$ has $f_* = \phi$ on each $A_i$ by
  (y).
\end{proof}

For a chain $d : \bigvee b \to K$ on $N$ events and $\phi \in \EN(B_a,B_b)$, write $d\cdot\phi :=
d \circ f$ for the $f$ with $f_* = \phi$.

\begin{definition}\label{def:artK}
  $\Art(K)$: $0$-generators the runs of $K$; $1$-generators the chains $\alpha$ of degree $1$, from
  $\alpha\cdot1$ to $\alpha\cdot s_t$, where $\cut(\alpha) = \{t\}$; $2$-generators the chains $e$
  of degree $2$, $\cut(e) = \{i < k\}$, $m := \cox(i,k)$, with
  \[
    \src(e) = \bigl(e\cdot\altw{s_i}{s_k}j\bigr)_{j<m}, \qquad \tgt(e) =
    \bigl(e\cdot\altw{s_k}{s_i}j\bigr)_{j<m} .
  \]
  For $f : K \to L$, $\Art(f) := \Ch(f)$ on generators. $\Br(K)$, $Q$, $W_K$, $\Br(f)$ are as in
  \texttt{def:W}, \texttt{prop:W}. $E_K(X) := Q(X)$, $E_K(\alpha) := Q(\top_\alpha)\,Q(\bot_\alpha)\inv$,
  where $\bot_\alpha$, $\top_\alpha$ are the refinements from $\alpha\cdot1$, $\alpha\cdot s_t$.
\end{definition}

\begin{theorem}[Theorem~1.1 of \texttt{mainv2}]\label{thm:main}
  Let $K$ be a bipointed precubical set.
  \begin{enumerate}
    \item $\Art(K)$ presents $\Br(K)$: $E_K : \pres{\Art(K)} \to \Br(K)$ is an equivalence.
    \item $\Art : \BPSet \to \Polygraph$ and $\Br : \BPSet \to \Cat$ are functors, and for every $f :
      K \to L$, $E_L \circ \pres{\Art(f)} = \Br(f) \circ E_K$.
    \item $\Art(Z) \cong \ArtBraid$ (\texttt{def:artin-poly}). Consequently $\Br(Z) \simeq
      \BraidPos$.
  \end{enumerate}
\end{theorem}
\begin{proof}
  Let $P^N_K := \BPSet(\bigvee -, K)$ on shapes on $N$ events, a presheaf on $\EN$ by
  \Cref{prop:cubes}. By \cite[Def.~7.4]{Z2}, $\Ch(K) = \coprod_N \Ch(K;N)$ with $\Ch(K;N)$ the slice
  over $K$, which is $\int P^N_K$; $W_K$ corresponds to $\coprod_N W_{P^N_K}$, as $f_* = 1$ means
  $\phi = 1$. A functor out of a coproduct is a family of functors, so \cite[Lem.~I.1.2]{GZ} gives
  $\Br(K) = \coprod_N \Br(P^N_K)$. Also $\Art(K) = \coprod_N \Art(P^N_K)$ (\Cref{def:art}), and $E_K$
  is $\coprod_N E_{P^N_K}$ (\Cref{thm:A}), since $\top_\alpha = \rho^\alpha_s$ and $\bot_\alpha =
  \rho^\alpha_1$.

  (i) \Cref{thm:A}. (ii) $f$ induces $P^N_f : P^N_K \to P^N_L$ by composition, functorially in $f$,
  with $\int P^N_f = \Ch(f)$; so $\Art(f) = \coprod_N \Art(P^N_f)$, $\Br(f) = \coprod_N
  \Br(P^N_f)$, and the square is the naturality in \Cref{thm:A}.

  (iii) $Z$ is terminal, so $P^N_Z = 1$, and by \Cref{cor:artin} sending $s_t \mapsto \sigma_t$ is a
  bijection on generators in each dimension, compatible with $\src$ and $\tgt$; it is an
  isomorphism by \texttt{prop:present-univ}(iii). Then $\Br(Z) \simeq \pres{\ArtBraid}$ by (i) and
  \texttt{lem:pres-functor}. Finally $\pres{\ArtBraid}$ has objects $N$, no arrows between distinct
  objects, and $\End(N)$ the free monoid on $\sigma_1,\dots,\sigma_{N-1}$, a path read left to right,
  modulo the congruence generated by Artin's relations; this is $B_N^+$ as in \texttt{mainv2}
  \cite{Artin}. So $\pres{\ArtBraid} \cong \BraidPos$.
\end{proof}
